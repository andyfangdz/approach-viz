# iOS Native App Implementation Plan

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a native iOS app (SwiftUI + RealityKit) for 3D instrument approach visualization, sharing the Rust core crate via UniFFI.

**Architecture:** SwiftUI drives UI (airport picker, approach selector, options). RealityView hosts a 3D scene with LowLevelMesh entities (terrain grid, approach path tubes, waypoint markers, runways). Rust core crate compiles to iOS via UniFFI for coordinate math. SQLite database (same as web app) bundles in the app for approach data.

**Tech Stack:** Swift 6, SwiftUI, RealityKit (iOS 18+), LowLevelMesh, CustomMaterial + Metal shaders, UniFFI, GRDB.swift, Rust (approach-viz-core crate)

**Spec:** `docs/superpowers/specs/2026-03-15-ios-native-app-design.md`

```rust
//! UniFFI exports for iOS.
//!
//! Mirrors `wasm.rs` — thin wrappers calling core logic with UniFFI annotations.

// UniFFI requires named return types (no bare tuples).

/// Local scene coordinate (x = east NM, z = -north NM).
#[derive(uniffi::Record)]
pub struct LocalCoord {
    pub x: f64,
    pub z: f64,
}

/// Projection scale factors at a latitude (NM per degree).
#[derive(uniffi::Record)]
pub struct ProjectionScales {
    pub east_nm_per_lon_deg: f64,
    pub north_nm_per_lat_deg: f64,
}

#[uniffi::export]
pub fn ios_lat_lon_to_local(lat: f64, lon: f64, ref_lat: f64, ref_lon: f64) -> LocalCoord {
    let (x, z) = crate::coords::lat_lon_to_local(lat, lon, ref_lat, ref_lon);
    LocalCoord { x, z }
}

#[uniffi::export]
pub fn ios_alt_to_y(alt_feet: f64, vertical_scale: f64) -> f64 {
    crate::coords::alt_to_y(alt_feet, vertical_scale)
}

#[uniffi::export]
pub fn ios_earth_curvature_drop_nm(x_nm: f64, z_nm: f64, ref_lat: f64) -> f64 {
    crate::coords::earth_curvature_drop_nm(x_nm, z_nm, ref_lat)
}

#[uniffi::export]
pub fn ios_geocentric_radius_nm(latitude_deg: f64) -> f64 {
    crate::coords::geocentric_radius_nm(latitude_deg)
}

#[uniffi::export]
pub fn ios_projection_scales_nm_per_degree(lat_deg: f64) -> ProjectionScales {
    let (east, north) = crate::coords::projection_scales_nm_per_degree(lat_deg);
    ProjectionScales {
        east_nm_per_lon_deg: east,
        north_nm_per_lat_deg: north,
    }
}

#[uniffi::export]
pub fn ios_normalize_heading(degrees: f64) -> f64 {
    crate::coords::normalize_heading(degrees)
}

#[uniffi::export]
pub fn ios_magnetic_to_true_heading(magnetic_course: f64, magnetic_variation: f64) -> f64 {
    crate::coords::magnetic_to_true_heading(magnetic_course, magnetic_variation)
}

uniffi::setup_scaffolding!();
```

- [ ] **Step 3: Create build-ios.sh**

Create `scripts/build-ios.sh`:

```bash
#!/usr/bin/env bash
set -euo pipefail

CRATE_DIR="crates/approach-viz-core"
OUT_DIR="ios/ApproachViz/RustBridge"
PROFILE="${1:-release}"
PROFILE_FLAG=""
TARGET_SUBDIR="release"

if [ "$PROFILE" = "debug" ]; then
    TARGET_SUBDIR="debug"
else
    PROFILE_FLAG="--release"
fi

echo "==> Building approach-viz-core for iOS ($PROFILE)..."

# Device (arm64)
cargo build -p approach-viz-core --target aarch64-apple-ios --features ios $PROFILE_FLAG

# Simulator (arm64 + x86_64)
cargo build -p approach-viz-core --target aarch64-apple-ios-sim --features ios $PROFILE_FLAG
cargo build -p approach-viz-core --target x86_64-apple-ios --features ios $PROFILE_FLAG

echo "==> Creating universal simulator binary..."
mkdir -p target/universal-ios-sim/$TARGET_SUBDIR
lipo -create \
    target/aarch64-apple-ios-sim/$TARGET_SUBDIR/libapproach_viz_core.a \
    target/x86_64-apple-ios/$TARGET_SUBDIR/libapproach_viz_core.a \
    -output target/universal-ios-sim/$TARGET_SUBDIR/libapproach_viz_core.a

echo "==> Generating Swift bindings..."
mkdir -p "$OUT_DIR"

# UniFFI proc-macro mode: build a host-target cdylib to extract metadata,
# then generate Swift bindings from it. iOS static libs (.a) don't embed
# the metadata that --library needs, so we build a macOS dylib for bindgen.
cargo build -p approach-viz-core --features ios
uniffi-bindgen generate \
    --library target/debug/libapproach_viz_core.dylib \
    --language swift \
    --out-dir "$OUT_DIR"

echo "==> Creating XCFramework..."
rm -rf ios/ApproachVizCore.xcframework
xcodebuild -create-xcframework \
    -library target/aarch64-apple-ios/$TARGET_SUBDIR/libapproach_viz_core.a \
    -headers "$OUT_DIR" \
    -library target/universal-ios-sim/$TARGET_SUBDIR/libapproach_viz_core.a \
    -headers "$OUT_DIR" \
    -output ios/ApproachVizCore.xcframework

echo "==> Done. XCFramework at ios/ApproachVizCore.xcframework"
echo "    Swift bindings at $OUT_DIR/"
```

Create `ios/ApproachViz/Services/DatabaseService.swift`:

```swift
import Foundation
import GRDB

final class DatabaseService: Sendable {
    private let dbPool: DatabasePool

    init(bundlePath: String) throws {
        var config = Configuration()
        config.readonly = true
        self.dbPool = try DatabasePool(path: bundlePath, configuration: config)
    }

    /// Convenience init loading from app bundle.
    convenience init() throws {
        guard let path = Bundle.main.path(forResource: "approach-viz", ofType: "sqlite") else {
            throw DatabaseError.databaseNotFound
        }
        try self.init(bundlePath: path)
    }

    func listAirports() throws -> [Airport] {
        try dbPool.read { db in
            try Row.fetchAll(db, sql: "SELECT id, name, lat, lon, elevation, mag_var FROM airports ORDER BY id")
                .map { row in
                    Airport(
                        id: row["id"],
                        name: row["name"],
                        lat: row["lat"],
                        lon: row["lon"],
                        elevation: row["elevation"],
                        magVar: row["mag_var"]
                    )
                }
        }
    }

    func listApproaches(airportId: String) throws -> [ApproachOption] {
        try dbPool.read { db in

    func loadWaypoints() throws -> [String: Waypoint] {

                """, arguments: [airportId])

        try dbPool.read { db in
            guard let row = try Row.fetchOne(db, sql: """
                SELECT airport_id, procedure_id, type, runway, data_json
                FROM approaches WHERE airport_id = ? AND procedure_id = ?
                """, arguments: [airportId, procedureId]) else {
                return nil
            }

            let dataJson: String = row["data_json"]
            guard let jsonData = dataJson.data(using: .utf8) else { return nil }

            let parsed = try JSONDecoder().decode(ApproachJsonData.self, from: jsonData)

            return ApproachData(
                airportId: row["airport_id"],
                procedureId: row["procedure_id"],
                type: row["type"],
                runway: row["runway"],
                transitions: parsed.transitions,
                finalLegs: parsed.finalLegs,
                missedLegs: parsed.missedLegs
            )
        }
    }

    func loadAirspace(lat: Double, lon: Double, radiusNm: Double) throws -> [AirspaceFeature] {
        // Approximate bounding box from NM radius
        let latDelta = radiusNm / 60.0
        let lonDelta = radiusNm / (60.0 * cos(lat * .pi / 180.0))

        return try dbPool.read { db in
            let rows = try Row.fetchAll(db, sql: """
                SELECT a.id, a.class, a.name, a.lower_alt, a.upper_alt, a.coordinates_json
                FROM airspace a
                INNER JOIN airspace_rtree r ON a.id = r.id
                WHERE r.min_lat <= ? AND r.max_lat >= ? AND r.min_lon <= ? AND r.max_lon >= ?
                """, arguments: [lat + latDelta, lat - latDelta, lon + lonDelta, lon - lonDelta])

            return rows.compactMap { row -> AirspaceFeature? in
                guard let coordsJson: String = row["coordinates_json"],
                      let coordsData = coordsJson.data(using: .utf8),
                      let coords = try? JSONDecoder().decode([[[Double]]].self, from: coordsData)
                else { return nil }

                return AirspaceFeature(
                    id: "\(row["id"] as Int)",
                    airspaceClass: row["class"],
                    name: row["name"],
                    lowerAlt: row["lower_alt"],
                    upperAlt: row["upper_alt"],
                    coordinates: coords
                )
            }
        }
    }

    enum DatabaseError: Error {
        case databaseNotFound
    }

/// Internal JSON shape matching the web app's `data_json` column.
/// The web app serializes Map entries as `[[key, [legs]], ...]` arrays.
private struct ApproachJsonData: Decodable {
    let transitions: [TransitionEntry]
    let finalLegs: [ApproachLeg]
    let missedLegs: [ApproachLeg]

    enum CodingKeys: String, CodingKey {
        case transitions, finalLegs, missedLegs
    }

    init(from decoder: Decoder) throws {
        let container = try decoder.container(keyedBy: CodingKeys.self)
        self.finalLegs = try container.decode([ApproachLeg].self, forKey: .finalLegs)
        self.missedLegs = try container.decode([ApproachLeg].self, forKey: .missedLegs)

        // Transitions are serialized as [[name, [legs]], ...] (JS Map.entries format)
        var transEntries: [TransitionEntry] = []
        var transArray = try container.nestedUnkeyedContainer(forKey: .transitions)
        while !transArray.isAtEnd {
            var pair = try transArray.nestedUnkeyedContainer()
            let name = try pair.decode(String.self)
            let legs = try pair.decode([ApproachLeg].self)
            transEntries.append(TransitionEntry(name: name, legs: legs))
        }
        self.transitions = transEntries
    }
}
```

```swift
import Foundation
import simd
import RealityKit

/// Orbital camera state: spherical coordinates around a target point.
struct CameraState {
    var target: SIMD3<Float> = .zero
    var distance: Float = 50        // NM from target
    var azimuth: Float = 0          // radians, 0 = looking from +Z toward origin
    var elevation: Float = 0.5      // radians above horizontal

    static let minDistance: Float = 2
    static let maxDistance: Float = 200
    static let minElevation: Float = 0.05
    static let maxElevation: Float = Float.pi / 2 - 0.05

    /// Camera world position computed from spherical coords.
    var cameraPosition: SIMD3<Float> {
        let cosElev = cos(elevation)
        let sinElev = sin(elevation)
        let cosAz = cos(azimuth)
        let sinAz = sin(azimuth)

        return target + SIMD3<Float>(
            distance * cosElev * sinAz,
            distance * sinElev,
            distance * cosElev * cosAz
        )
    }

    /// 4×4 view transform (look-at from cameraPosition toward target, Y-up).
    var viewTransform: float4x4 {
        let pos = cameraPosition
        let forward = normalize(target - pos)
        let right = normalize(cross(forward, SIMD3<Float>(0, 1, 0)))
        let up = cross(right, forward)

        var m = float4x4(1)
        m.columns.0 = SIMD4(right.x, up.x, -forward.x, 0)
        m.columns.1 = SIMD4(right.y, up.y, -forward.y, 0)
        m.columns.2 = SIMD4(right.z, up.z, -forward.z, 0)
        m.columns.3 = SIMD4(pos.x, pos.y, pos.z, 1)
        return m
    }

    /// Orbit: rotate azimuth/elevation by delta.
    mutating func orbit(deltaAzimuth: Float, deltaElevation: Float) {
        azimuth += deltaAzimuth
        elevation = min(max(elevation + deltaElevation, Self.minElevation), Self.maxElevation)
    }

    /// Zoom: multiply distance by factor.
    mutating func zoom(by factor: Float) {
        distance = min(max(distance * factor, Self.minDistance), Self.maxDistance)
    }

    /// Pan: translate target in the camera's local XZ plane.
    mutating func pan(deltaRight: Float, deltaUp: Float) {
        let forward = normalize(target - cameraPosition)
        let right = normalize(cross(forward, SIMD3<Float>(0, 1, 0)))
        let up = cross(right, forward)

        target += right * deltaRight + up * deltaUp
    }
}
```

```swift
import Foundation
import UIKit

actor TileService {
    private let cache = NSCache<NSString, NSData>()
    private let session: URLSession

    static let tileBaseURL = "https://elevation-tiles-prod.s3.amazonaws.com/terrarium"
    static let tileZoom = 10
    static let tileSize = 256

    init() {
        let config = URLSessionConfiguration.default
        config.urlCache = URLCache(
            memoryCapacity: 50 * 1024 * 1024,   // 50 MB memory
            diskCapacity: 200 * 1024 * 1024      // 200 MB disk
        )
        self.session = URLSession(configuration: config)
    }

    /// Decode Terrarium elevation from RGB pixel values.
    /// Formula: elevation_meters = r * 256 + g + b / 256 - 32768
    static func decodeTerrarium(r: UInt8, g: UInt8, b: UInt8) -> Double {
        Double(r) * 256.0 + Double(g) + Double(b) / 256.0 - 32768.0
    }

    /// Lon → tile X at given zoom (floating point, no floor).
    static func lonToTileXFloat(_ lon: Double, zoom: Int) -> Double {
        ((lon + 180.0) / 360.0) * pow(2.0, Double(zoom))
    }

    /// Lat → tile Y at given zoom (floating point, no floor).
    static func latToTileYFloat(_ lat: Double, zoom: Int) -> Double {
        let latRad = lat * .pi / 180.0
        (1.0 - log(tan(.pi / 4.0 + latRad / 2.0)) / .pi) * 0.5 * pow(2.0, Double(zoom))
    }

    /// Fetch elevation grid for a bounding box, returning positions in local NM coordinates.
    func loadTerrainGrid(
        refLat: Double, refLon: Double,
        radiusNm: Double,
        gridSegments: Int = 140
    ) async throws -> TerrainData {
        let latDelta = radiusNm / 60.0
        let lonDelta = radiusNm / (60.0 * cos(refLat * .pi / 180.0))
        let minLat = refLat - latDelta
        let maxLat = refLat + latDelta
        let minLon = refLon - lonDelta
        let maxLon = refLon + lonDelta

        // Determine tiles needed
        let zoom = Self.tileZoom
        let minTileX = Int(Self.lonToTileXFloat(minLon, zoom: zoom))
        let maxTileX = Int(Self.lonToTileXFloat(maxLon, zoom: zoom))
        let minTileY = Int(Self.latToTileYFloat(maxLat, zoom: zoom)) // note: Y is inverted
        let maxTileY = Int(Self.latToTileYFloat(minLat, zoom: zoom))

        let tilesWide = maxTileX - minTileX + 1
        let tilesTall = maxTileY - minTileY + 1
        let pixelWidth = tilesWide * Self.tileSize
        let pixelHeight = tilesTall * Self.tileSize

        // Fetch all tiles in parallel
        var pixelData = [UInt8](repeating: 0, count: pixelWidth * pixelHeight * 4)

        try await withThrowingTaskGroup(of: (Int, Int, Data).self) { group in
            for ty in minTileY...maxTileY {
                for tx in minTileX...maxTileX {
                    group.addTask {
                        let data = try await self.fetchTile(x: tx, y: ty, zoom: zoom)
                        return (tx - minTileX, ty - minTileY, data)
                    }
                }
            }

            for try await (offsetX, offsetY, tileData) in group {
                // Decode PNG to raw RGBA pixels and copy into composite
                guard let image = UIImage(data: tileData),
                      let cgImage = image.cgImage else { continue }

                let w = Self.tileSize
                let h = Self.tileSize
                let colorSpace = CGColorSpaceCreateDeviceRGB()
                var tilePixels = [UInt8](repeating: 0, count: w * h * 4)
                let context = CGContext(
                    data: &tilePixels, width: w, height: h,
                    bitsPerComponent: 8, bytesPerRow: w * 4,
                    space: colorSpace,
                    bitmapInfo: CGImageAlphaInfo.premultipliedLast.rawValue
                )
                context?.draw(cgImage, in: CGRect(x: 0, y: 0, width: w, height: h))

                // Copy tile pixels into composite at correct offset
                let destX = offsetX * w
                let destY = offsetY * h
                for row in 0..<h {
                    let srcStart = row * w * 4
                    let dstStart = ((destY + row) * pixelWidth + destX) * 4
                    pixelData.replaceSubrange(dstStart..<(dstStart + w * 4),
                                              with: tilePixels[srcStart..<(srcStart + w * 4)])
                }
            }
        }

        // Build vertex grid
        let verticesPerRow = gridSegments + 1
        let totalVertices = verticesPerRow * verticesPerRow
        var positions = [Float](repeating: 0, count: totalVertices * 3)

        let altitudeScale: Double = 1.0 / 6076.12
        let metersToFeet: Double = 3.28084

        for row in 0..<verticesPerRow {
            for col in 0..<verticesPerRow {
                let lat = maxLat - Double(row) / Double(gridSegments) * (maxLat - minLat)
                let lon = minLon + Double(col) / Double(gridSegments) * (maxLon - minLon)

                // Sample pixel
                let tileXF = Self.lonToTileXFloat(lon, zoom: zoom) - Double(minTileX)
                let tileYF = Self.latToTileYFloat(lat, zoom: zoom) - Double(minTileY)
                let px = min(max(Int(tileXF * Double(Self.tileSize)), 0), pixelWidth - 1)
                let py = min(max(Int(tileYF * Double(Self.tileSize)), 0), pixelHeight - 1)

                let pixelIdx = (py * pixelWidth + px) * 4
                let r = pixelData[pixelIdx]
                let g = pixelData[pixelIdx + 1]
                let b = pixelData[pixelIdx + 2]

                let elevMeters = Self.decodeTerrarium(r: r, g: g, b: b)
                let elevFeet = elevMeters * metersToFeet

                // Local coordinates (using simplified projection for terrain grid)
                let cosRef = cos(refLat * .pi / 180.0)
                let x = Float((lon - refLon) * 60.0 * cosRef)
                let z = Float(-(lat - refLat) * 60.0)
                let y = Float(elevFeet * altitudeScale)

                let idx = (row * verticesPerRow + col) * 3
                positions[idx] = x
                positions[idx + 1] = y
                positions[idx + 2] = z
            }
        }

        return TerrainData(
            gridWidth: verticesPerRow,
            gridHeight: verticesPerRow,
            positions: positions,
            minLat: minLat, maxLat: maxLat,
            minLon: minLon, maxLon: maxLon
        )
    }

    private func fetchTile(x: Int, y: Int, zoom: Int) async throws -> Data {
        let key = NSString(string: "\(zoom)/\(x)/\(y)")
        if let cached = cache.object(forKey: key) {
            return cached as Data
        }

        let url = URL(string: "\(Self.tileBaseURL)/\(zoom)/\(x)/\(y).png")!
        let (data, _) = try await session.data(from: url)
        cache.setObject(data as NSData, forKey: key)
        return data
    }
}
```

```metal
// Barycentric wireframe shader.
// Vertex attribute layout: position (float3), normal (float3), barycentric (float3)

[[visible]]
void terrainWireframeSurface(realitykit::surface_parameters params)
{
    // Barycentric coordinates passed as custom UV
    float3 bary = params.geometry().uv1();

    // Wireframe edge detection: find minimum barycentric coord
    float minBary = min(bary.x, min(bary.y, bary.z));

    // Anti-aliased edge width
    float edgeWidth = 0.02;
    float edge = smoothstep(0.0, edgeWidth, minBary);

    // Terrain wireframe color: #4ea0db at 58% opacity
    half3 wireColor = half3(0.306, 0.627, 0.859);
    half3 fillColor = half3(0.047, 0.102, 0.184); // #0c1a2f

    half3 color = mix(wireColor, fillColor, half(edge));
    half alpha = mix(half(0.58), half(0.12), half(edge));

    params.surface().set_base_color(color);
    params.surface().set_opacity(alpha);
    params.surface().set_metallic(0.0);
    params.surface().set_roughness(0.8);
}
```

```swift
import RealityKit
import Metal

/// Builds a terrain wireframe grid from elevation data using LowLevelMesh.
enum TerrainEntity {

    /// Build a terrain entity from TerrainData.
    @MainActor
    static func build(from data: TerrainData, verticalScale: Float) throws -> ModelEntity {
        let w = data.gridWidth
        let h = data.gridHeight
        let vertexCount = w * h
        let quadCount = (w - 1) * (h - 1)
        let triangleCount = quadCount * 2
        let indexCount = triangleCount * 3

        // Define vertex layout: position (float3) + barycentric (float3)
        var descriptor = LowLevelMesh.Descriptor()
        descriptor.vertexCapacity = vertexCount
        descriptor.indexCapacity = indexCount

        // Vertex attributes
        let stride = MemoryLayout<SIMD3<Float>>.stride * 2 // pos + bary = 24 bytes
        descriptor.vertexAttributes = [
            .init(semantic: .position, format: .float3, layoutIndex: 0, offset: 0),
            .init(semantic: .uv1, format: .float3, layoutIndex: 0, offset: MemoryLayout<SIMD3<Float>>.stride),
        ]
        descriptor.vertexLayouts = [
            .init(bufferStride: stride),
        ]
        descriptor.indexType = .uint32

        let mesh = try LowLevelMesh(descriptor: descriptor)

        // Fill vertex buffer
        mesh.withUnsafeMutableBytes(bufferIndex: 0) { buffer in
            let vertices = buffer.bindMemory(to: (SIMD3<Float>, SIMD3<Float>).self)
            for row in 0..<h {
                for col in 0..<w {
                    let srcIdx = (row * w + col) * 3
                    let x = data.positions[srcIdx]
                    let y = data.positions[srcIdx + 1] * verticalScale
                    let z = data.positions[srcIdx + 2]
                    let position = SIMD3<Float>(x, y, z)

                    // Barycentric coords for wireframe — alternate per triangle vertex
                    // Assign based on position in quad to ensure each triangle gets (1,0,0), (0,1,0), (0,0,1)
                    let baryIndex = (row + col) % 3
                    let bary: SIMD3<Float>
                    switch baryIndex {
                    case 0: bary = SIMD3<Float>(1, 0, 0)
                    case 1: bary = SIMD3<Float>(0, 1, 0)
                    default: bary = SIMD3<Float>(0, 0, 1)
                    }

                    vertices[row * w + col] = (position, bary)
                }
            }
        }

        // Fill index buffer
        mesh.withUnsafeMutableIndices { buffer in
            let indices = buffer.bindMemory(to: UInt32.self)
            var idx = 0
            for row in 0..<(h - 1) {
                for col in 0..<(w - 1) {
                    let a = UInt32(row * w + col)
                    let b = a + 1
                    let c = UInt32((row + 1) * w + col)
                    let d = c + 1

                    // Two CCW triangles per quad
                    indices[idx] = a; indices[idx+1] = c; indices[idx+2] = b; idx += 3
                    indices[idx] = b; indices[idx+1] = c; indices[idx+2] = d; idx += 3
                }
            }
        }

        // Set mesh part
        let part = LowLevelMesh.Part(
            indexOffset: 0,
            indexCount: indexCount,
            topology: .triangle,
            materialIndex: 0
        )
        mesh.parts.replaceAll([part])

        // Create material with wireframe shader
        var material = CustomMaterial(surfaceShader: .init(
            named: "terrainWireframeSurface",
            in: MetalLibLoader.library
        ), lightingModel: .unlit)
        material.faceCulling = .none

        return ModelEntity(mesh: try .init(from: mesh), materials: [material])
    }
}

/// Helper to load the default Metal library.
enum MetalLibLoader {
    static let library: MTLLibrary = {
        let device = MTLCreateSystemDefaultDevice()!
        return device.makeDefaultLibrary()!
    }()
}
```

```swift
import Foundation
import simd

/// Result of path building: 3D points + vertical guide lines.
struct PathBuildResult {
    let points: [SIMD3<Float>]
    let verticalLines: [SIMD3<Float>]  // base positions for altitude guide lines
}

/// Builds 3D polyline points from approach legs.
///
/// Ports the core algorithm from `app/scene/approach-path/path-builder.ts`.
/// MVP handles: IF, TF, DF, CF fix-join legs. Future: RF arcs, heading legs (CA, VI), holds.
enum ApproachPathBuilder {

    static let altitudeScale: Double = 1.0 / 6076.12

    static func build(
        legs: [ApproachLeg],
        waypoints: [String: Waypoint],
        altitudes: [Double],
        refLat: Double, refLon: Double,
        magVar: Double,
        verticalScale: Double
    ) -> PathBuildResult {
        var points: [SIMD3<Float>] = []
        var verticalLines: [SIMD3<Float>] = []

        for (i, leg) in legs.enumerated() {
            let alt = i < altitudes.count ? altitudes[i] : 0

            // Resolve waypoint position
            guard let wp = resolveWaypoint(waypoints: waypoints, id: leg.waypointId) else {
                continue
            }

            // Convert to local coordinates via Rust UniFFI (or Swift fallback)
            let local = latLonToLocal(lat: wp.lat, lon: wp.lon, refLat: refLat, refLon: refLon)
            let y = Float(alt * altitudeScale * verticalScale)
            let point = SIMD3<Float>(Float(local.x), y, Float(local.z))

            // Avoid duplicate points
            if let last = points.last, simd_distance(last, point) < 1e-6 {
                continue
            }

            points.append(point)
            verticalLines.append(SIMD3<Float>(Float(local.x), 0, Float(local.z)))
        }

        return PathBuildResult(points: points, verticalLines: verticalLines)
    }

    /// Resolve waypoint by ID, with fallback stripping prefix.
    /// Matches web app's `resolveWaypoint()` in coordinates.ts.
    private static func resolveWaypoint(waypoints: [String: Waypoint], id: String) -> Waypoint? {
        if let wp = waypoints[id] { return wp }
        // Fallback: try last component after underscore
        if let lastPart = id.split(separator: "_").last {
            return waypoints[String(lastPart)]
        }
        return nil
    }

    /// Local coordinate conversion.
    /// Uses Rust UniFFI if available, otherwise pure Swift fallback.
    /// Internal access: also used by RunwayEntity and other scene components.
    static func latLonToLocal(
        lat: Double, lon: Double, refLat: Double, refLon: Double
    ) -> (x: Double, z: Double) {
        // Once XCFramework is linked, replace with:
        // let coord = iosLatLonToLocal(lat: lat, lon: lon, refLat: refLat, refLon: refLon)
        // return (coord.x, coord.z)

        #warning("Remove Swift fallback once UniFFI XCFramework is linked")
        // Swift fallback (same WGS84 math as Rust core)
        let degToRad = Double.pi / 180.0
        let metersToNm = 1.0 / 1852.0
        let a = 6_378_137.0
        let f = 1.0 / 298.257_223_563
        let e2 = f * (2.0 - f)

        let phi = refLat * degToRad
        let sinPhi = sin(phi)
        let cosPhi = cos(phi)
        let denom = sqrt(1.0 - e2 * sinPhi * sinPhi)
        let N = a / denom
        let M = a * (1.0 - e2) / (denom * denom * denom)

        let dLatRad = (lat - refLat) * degToRad
        let dLonRad = (lon - refLon) * degToRad

        let x = dLonRad * N * cosPhi * metersToNm
        let z = -(dLatRad * M * metersToNm)
        return (x, z)
    }
}
```

Note: The Swift `latLonToLocal` fallback is included so the path builder works before the UniFFI bridge is integrated into Xcode. Once the XCFramework is linked, uncomment the UniFFI call and remove the fallback.

```swift
import RealityKit
import simd

/// Segment types with their colors (matching web app).
enum ApproachSegmentType {
    case approach
    case transition
    case missed
    case hold

    var color: SIMD3<Float> {
        switch self {
        case .approach:  return SIMD3<Float>(0.2, 0.8, 0.2)   // green
        case .transition: return SIMD3<Float>(0.3, 0.5, 1.0)  // blue
        case .missed:    return SIMD3<Float>(1.0, 0.3, 0.3)   // red
        case .hold:      return SIMD3<Float>(1.0, 0.9, 0.2)   // yellow
        }
    }
}

/// Builds a tube mesh along a 3D polyline using LowLevelMesh.
enum ApproachPathEntity {

    static let tubeRadius: Float = 0.08
    static let radialSegments = 8

    /// Build a tube entity from a polyline of 3D points.
    @MainActor
    static func build(
        points: [SIMD3<Float>],
        segmentType: ApproachSegmentType
    ) throws -> ModelEntity? {
        guard points.count >= 2 else { return nil }

        let color = segmentType.color
        let segments = points.count - 1
        let vertsPerRing = radialSegments + 1 // duplicate first vertex for UV seam
        let vertexCount = (segments + 1) * vertsPerRing
        let indexCount = segments * radialSegments * 6 // 2 triangles per quad

        // Vertex layout: position (float3) + color (float3 via uv0)
        let vertexStride = MemoryLayout<SIMD3<Float>>.stride * 2 // 24 bytes
        var descriptor = LowLevelMesh.Descriptor()
        descriptor.vertexCapacity = vertexCount
        descriptor.indexCapacity = indexCount
        descriptor.vertexAttributes = [
            .init(semantic: .position, format: .float3, layoutIndex: 0, offset: 0),
            .init(semantic: .uv0, format: .float3, layoutIndex: 0,
                  offset: MemoryLayout<SIMD3<Float>>.stride),
        ]
        descriptor.vertexLayouts = [.init(bufferStride: vertexStride)]
        descriptor.indexType = .uint32

        let mesh = try LowLevelMesh(descriptor: descriptor)

        // Fill vertices: ring of points at each path position
        mesh.withUnsafeMutableBytes(bufferIndex: 0) { buffer in
            let verts = buffer.bindMemory(to: (SIMD3<Float>, SIMD3<Float>).self)
            var vi = 0

            for i in 0...segments {
                let center = points[min(i, points.count - 1)]

                // Compute tangent direction
                let tangent: SIMD3<Float>
                if i == 0 {
                    tangent = normalize(points[1] - points[0])
                } else if i == segments {
                    tangent = normalize(points[segments] - points[segments - 1])
                } else {
                    tangent = normalize(points[i + 1] - points[i - 1])
                }

                // Build orthonormal basis for the ring
                let up = SIMD3<Float>(0, 1, 0)
                var right = normalize(cross(tangent, up))
                if length(right) < 0.001 {
                    right = normalize(cross(tangent, SIMD3<Float>(1, 0, 0)))
                }
                let ringUp = cross(right, tangent)

                // Generate ring vertices
                for j in 0...radialSegments {
                    let angle = Float(j) / Float(radialSegments) * 2.0 * .pi
                    let offset = right * cos(angle) * tubeRadius + ringUp * sin(angle) * tubeRadius
                    verts[vi] = (center + offset, color)
                    vi += 1
                }
            }
        }

        // Fill indices
        mesh.withUnsafeMutableIndices { buffer in
            let indices = buffer.bindMemory(to: UInt32.self)
            var idx = 0
            for i in 0..<segments {
                for j in 0..<radialSegments {
                    let a = UInt32(i * vertsPerRing + j)
                    let b = a + 1
                    let c = UInt32((i + 1) * vertsPerRing + j)
                    let d = c + 1

                    indices[idx] = a; indices[idx+1] = c; indices[idx+2] = b; idx += 3
                    indices[idx] = b; indices[idx+1] = c; indices[idx+2] = d; idx += 3
                }
            }
        }

        let part = LowLevelMesh.Part(
            indexOffset: 0, indexCount: indexCount,
            topology: .triangle, materialIndex: 0
        )
        mesh.parts.replaceAll([part])

        var material = CustomMaterial(surfaceShader: .init(
            named: "vertexColorSurface",
            in: MetalLibLoader.library
        ), lightingModel: .lit)
        material.faceCulling = .back

        return ModelEntity(mesh: try .init(from: mesh), materials: [material])
    }
}
```

```swift
            }
```

```swift
import SwiftUI

struct AirportPicker: View {
    let airports: [Airport]
    @Binding var selected: Airport?
    @State private var searchText = ""

    private var filtered: [Airport] {
        if searchText.isEmpty { return Array(airports.prefix(50)) }
        let query = searchText.uppercased()
        return airports.filter {
            $0.id.contains(query) || $0.name.uppercased().contains(query)
        }.prefix(50).map { $0 }
    }

    var body: some View {
        VStack(spacing: 0) {
            HStack {
                Image(systemName: "magnifyingglass")
                    .foregroundStyle(.secondary)
                TextField("Search airport (ICAO or name)", text: $searchText)
                    .textInputAutocapitalization(.characters)
                    .autocorrectionDisabled()
                if !searchText.isEmpty {
                    Button { searchText = "" } label: {
                        Image(systemName: "xmark.circle.fill")
                            .foregroundStyle(.secondary)
                    }
                }
            }
            .padding(10)
            .background(.ultraThinMaterial)

            if selected == nil || !searchText.isEmpty {
                List(filtered) { airport in
                    Button {
                        selected = airport
                        searchText = ""
                    } label: {
                        HStack {
                            Text(airport.id).bold().monospaced()
```

```swift
                VStack(spacing: 8) {

                        .frame(maxHeight: state.selectedAirport == nil ? 300 : 44)

                    if !approaches.isEmpty {
                        ApproachSelector(approaches: approaches, selected: $state.selectedApproach)
                    }
                }
                .background(.ultraThinMaterial)

                Spacer()

                // Bottom: options toggle
                HStack {
                    Spacer()
                    Button {
                        withAnimation { showOptions.toggle() }
                    } label: {
                        Image(systemName: "slider.horizontal.3")
                            .padding(12)
                            .background(.ultraThinMaterial)
                            .clipShape(Circle())
                    }
                    .padding()
                }

                if showOptions {
                    OptionsPanel(state: state)
                        .padding(.horizontal)
                        .padding(.bottom)
                        .transition(.move(edge: .bottom))
                }
            }

            // Loading indicator
            if state.isLoading {
                ProgressView()
                    .scaleEffect(1.5)
                    .frame(maxWidth: .infinity, maxHeight: .infinity)
                    .background(.black.opacity(0.3))
            }
        }
        .task {
            // Initialize database
            do {
                let database = try DatabaseService()
                self.db = database
                self.loader = SceneDataLoader(db: database)
                self.airports = try database.listAirports()
            } catch {
                state.loadError = "Failed to load database: \(error.localizedDescription)"
            }
        }
        .onChange(of: state.selectedAirport) { _, newAirport in
            guard let airport = newAirport, let db else { return }
            approaches = (try? db.listApproaches(airportId: airport.id)) ?? []
            state.selectedApproach = approaches.first
        }
        .onChange(of: state.selectedApproach) { _, newApproach in
            guard let airport = state.selectedAirport,
                  let approach = newApproach,
                  let loader else { return }

            Task {
                await loader.loadScene(
                    airportId: airport.id,
                    procedureId: approach.procedureId,
                    state: state
                )

                // Load terrain
                let terrain = try? await tileService.loadTerrainGrid(
                    refLat: airport.lat,
                    refLon: airport.lon,
                    radiusNm: Double(state.terrainRadiusNm)
                )
                state.terrainData = terrain
            }
        }
    }
}
```

```swift
    func latLonAtRef() {
        let coord = iosLatLonToLocal(lat: 40.64, lon: -73.78, refLat: 40.64, refLon: -73.78)
        #expect(abs(coord.x) < 1e-10)
        #expect(abs(coord.z) < 1e-10)
    }

    @Test("lat_lon_to_local one degree north ≈ 60 NM")
    func latLonOneDegNorth() {
        let coord = iosLatLonToLocal(lat: 41.64, lon: -73.78, refLat: 40.64, refLon: -73.78)
        // z is -north, so z ≈ -60
        #expect(abs(coord.z - (-60.0)) < 1.0)
    }

    @Test("alt_to_y scales correctly")
    func altToY() {
        let y = iosAltToY(altFeet: 1000, verticalScale: 1.0)
        let expected = 1000.0 / 6076.12
        #expect(abs(y - expected) < 1e-6)
    }

    @Test("earth_curvature_drop at 60 NM ≈ 0.52 NM")
    func curvatureDrop() {
        let drop = iosEarthCurvatureDropNm(xNm: 60.0, zNm: 0.0, refLat: 40.0)
        #expect(abs(drop - 0.52) < 0.05)
    }

    @Test("geocentric_radius at equator ≈ 3443.9 NM")
    func geocentricRadius() {
        let r = iosGeocentricRadiusNm(latitudeDeg: 0.0)
        #expect(abs(r - 3443.9) < 0.1)
    }

}
```
